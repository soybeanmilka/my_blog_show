差分数组 (Difference Array)

1 核心思想

差分数组是前缀和的逆运算，适用于频繁对数组某个区间进行增减操作的场景。

  • 暴力做法：每次区间修改 O(n)，m 次操作 O(n*m)
  • 差分数组：每次修改 O(1)，最后还原 O(n)，总复杂度 O(n + m)

2 一、基本原理

2.1 构造差分数组

1

diff[i] = nums[i] - nums[i-1]

2.2 区间增减操作

对区间 [i, j] 的所有元素 +val：

1
2

diff[i] += val;
diff[j + 1] -= val;   // 注意边界：j+1 不能越界

原理：

  • diff[i] += val → 从 i 开始，后面所有元素都加了 val
  • diff[j+1] -= val → 从 j+1 开始，抵消前面的增量
  • 综合效果：只有 [i, j] 区间加了 val

2.3 还原结果数组

对 diff 做一次前缀和，得到最终的 nums：

1
2
3
4

res[0] = diff[0];
for (int i = 1; i < n; i++) {
    res[i] = res[i - 1] + diff[i];
}

class Difference {
private:
    vector<int> diff;

public:
    // 构造差分数组
    Difference(vector<int>& nums) {
        diff.resize(nums.size());
        diff[0] = nums[0];
        for (int i = 1; i < nums.size(); i++) {
            diff[i] = nums[i] - nums[i - 1];
        }
    }

    // 区间 [i, j] 增加 val（可为负数）
    void increment(int i, int j, int val) {
        diff[i] += val;
        if (j + 1 < diff.size()) {
            diff[j + 1] -= val;
        }
        // j+1 越界说明加到末尾，不用减
    }

    // 返回最终结果数组
    vector<int> result() {
        vector<int> res(diff.size());
        res[0] = diff[0];
        for (int i = 1; i < diff.size(); i++) {
            res[i] = res[i - 1] + diff[i];
        }
        return res;
    }
};

vector<int> getModifiedArray(int n, vector<vector<int>>& updates) {
    vector<int> diff(n, 0);
    for (auto& op : updates) {
        int s = op[0], e = op[1], val = op[2];
        diff[s] += val;
        if (e + 1 < n) diff[e + 1] -= val;
    }
    // 还原
    vector<int> res(n);
    res[0] = diff[0];
    for (int i = 1; i < n; i++) {
        res[i] = res[i - 1] + diff[i];
    }
    return res;
}

vector<int> corpFlightBookings(vector<vector<int>>& bookings, int n) {
    vector<int> diff(n, 0);
    for (auto& b : bookings) {
        int first = b[0] - 1;   // 注意题目是 1-based
        int last = b[1] - 1;
        int seats = b[2];
        diff[first] += seats;
        if (last + 1 < n) diff[last + 1] -= seats;
    }
    // 前缀和还原
    for (int i = 1; i < n; i++) {
        diff[i] += diff[i - 1];
    }
    return diff;
}

bool carPooling(vector<vector<int>>& trips, int capacity) {
    vector<int> diff(1001, 0);  // 站点最多 1000
    for (auto& t : trips) {
        int num = t[0], from = t[1], to = t[2];
        diff[from] += num;
        diff[to] -= num;   // 到 to 站下车，不用 to+1
    }

    int cur = 0;
    for (int x : diff) {
        cur += x;
        if (cur > capacity) return false;
    }
    return true;
}

7 六、解题技巧

  1. 判断要不要用差分： 题目出现「多次区间加减，最后求结果」，直接想差分
  2. 注意下标： 题目给的是 1-based 还是 0-based，要转换
  3. 边界处理： j+1 越界时不用减，说明加到末尾
  4. 不需要还原整个数组： 比如拼车问题，边累加边判断即可
  5. 数据范围大时： 用 map 存离散的差分点（扫描线思想）
